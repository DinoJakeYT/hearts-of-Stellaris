### The Colossus picks its targets on the map

hos_colossus_category = {

	hos_colossus_fire = {
		icon = GFX_decision_hos_colossus_fire
		state_target = yes
		on_map_mode = map_only
		target_root_trigger = {
			has_country_flag = hos_colossus_built
		}
		target_trigger = {
			FROM = {
				NOT = { has_state_flag = hos_colossus_hit }
				controller = { has_war_with = ROOT }
			}
		}
		available = {
			NOT = { has_country_flag = hos_colossus_charging }
		}
		fire_only_once = no
		cost = 100
		complete_effect = {
			save_global_event_target_as = hos_colossus_user
			FROM = {
				save_global_event_target_as = hos_colossus_target
				OWNER = { save_global_event_target_as = hos_colossus_victim }
			}
			if = {
				limit = { has_country_flag = hos_colossus_world_cracker }
				hos_world_cracker = yes
			}
			else_if = {
				limit = { has_country_flag = hos_colossus_neutron_sweep }
				hos_neutron_sweep = yes
			}
			else = {
				hos_nanobot_diffuser = yes
			}
		}
		ai_will_do = {
			base = 1
			modifier = {
				factor = 20
				FROM = { is_capital = yes }
			}
			modifier = {
				factor = 0
				has_idea = hos_ethic_pacifist
			}
		}
	}
}
