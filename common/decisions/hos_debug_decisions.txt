		icon = GFX_decision_hos_defender
		visible = {

			NOT = { has_global_flag = hos_crisis_arrived }

		}

		visible = {
			is_debug = yes
			NOT = { has_country_flag = hos_colossus_built }
		}
		complete_effect = {
			set_country_flag = hos_satellite
			set_country_flag = hos_spaceport
			set_country_flag = hos_moon

			base = 0
		}
