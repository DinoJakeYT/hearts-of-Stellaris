	title = hos_anomaly.1.t
	desc = hos_anomaly.1.d
	picture = GFX_report_event_hos_anom_whale

	immediate = { hidden_effect = { set_country_flag = hos_anom_whale } }

		name = hos_anomaly.1.a

		name = hos_anomaly.1.b

		add_tech_bonus = { name = hos_anomaly.1.t bonus = 1 uses = 1 category = industry }
		add_stability = -0.03
	}
}
