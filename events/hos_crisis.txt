# The crisis declares war on the world. ROOT is the crisis.
country_event = {
	id = hos_crisis.20
	hidden = yes
	is_triggered_only = yes

	immediate = {
		every_other_country = {
			limit = {
				NOT = { hos_is_crisis_country = yes }
				NOT = { has_war_with = ROOT }
			}
			set_country_flag = hos_fought_crisis
			ROOT = { declare_war_on = { target = PREV type = annex_everything } }
		}
	}

	option = { name = hos_crisis.20.a }
}

# The Contingency subverts a machine nation's robots
country_event = {
	id = hos_crisis.21
	title = hos_crisis.21.t
	desc = hos_crisis.21.d
	picture = GFX_report_event_hos_synth_betrayal
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			random_owned_controlled_state = {
				limit = {
					is_capital = no
					NOT = { has_state_flag = hos_crisis_state }
				}
				set_state_flag = hos_crisis_state
				CTG = { transfer_state = PREV }
			}
		}
	}

	option = {
		name = hos_crisis.21.a
		add_timed_idea = { idea = hos_synth_betrayal days = 180 }
	}
}
